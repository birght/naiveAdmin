<route lang="json">
{
  "name": "scbagldetail",
  "meta": {
    "layout": "dashboard"
  }
}
</route>
<template>
  <n-backtop target=".hzp_container" />
  <div class="h-full hzp_container" ref="hzpContainer">
    <div class="hzp_progessbar" :style="{ width: scrollProgress + '%' }"></div>
    <div class="hzp_header">
      <div v-for="item in tagList" :key="item.id">
        <n-badge v-if="item.messagesNum !== null" :value="item.messagesNum">
          <div
            class="flex px-3 cursor-pointer indent-xs border-b-fuchsia"
            :class="{ ' shi_flowing_text': selectedTagId === item.id }"
          >
            <div @click="checkTagEmits(item)" class="text-center">
              {{ item.moduleName }}
            </div>
          </div>
        </n-badge>
        <div
          v-else
          class="flex px-3 cursor-pointer indent-xs border-b-fuchsia"
          :class="{ ' shi_flowing_text': selectedTagId === item.id }"
        >
          <div @click="checkTagEmits(item)" class="text-center">
            {{ item.moduleName }}
          </div>
        </div>
      </div>
    </div>
    <div v-if="isLoading" class="hzp_content" animated>
      <n-skeleton :rows="15" animated />
    </div>
    <!-- 新的内容区域 -->
    <div class="p-4 hzp_content" v-else>
      <div v-for="(section, index) in exampleData" :key="index" class="flex w-full">
        <div v-for="item in section" :key="item.itemPid" class="w-full">
          <div v-if="item.showType === 'title'" class="mt-4">
            <div class="my-3 shititle">
              {{ item.showName }}
            </div>
          </div>
          <div v-if="item.showType === 'radio'" class="shiradio_cc">
            <div :class="item.itemValue === '1' ? 'shiradio_check' : 'shiradio_uncheck'"></div>
            <div class="mt-3 align-middle shiradio1">
              {{ item.showName }}
              <n-popover
                v-if="item.resultType === '1' || item.resultType === '3'"
                placement="right"
                :width="200"
                trigger="hover"
                :content="item.message || ''"
              >
                <template #reference>
                  <n-icon class="text-bluegray"><InfoFilled /></n-icon>
                </template>
              </n-popover>
            </div>
          </div>
          <div v-if="item.showType === 'card'" class="flex">
            <div
              v-for="(leve2item, index2) in item.cardList"
              :key="leve2item.itemPid"
              class="box-border w-1/3 p-2"
            >
              <n-card :class="'markcard' + ((index2 % 3) + 1)">
                <div :class="'shi_card_bg' + ((index2 % 3) + 1)">
                  <strong>{{ leve2item.showName }}</strong>
                  <div class="shi_xuline"></div>
                  <div>
                    {{ leve2item.itemValue }}
                    <n-popover
                      v-if="leve2item.resultType === '2'"
                      placement="right"
                      :width="200"
                      trigger="hover"
                      :content="item.message"
                    >
                      <template #reference>
                        <n-icon class="mx-1 text-red-4"><InfoFilled /></n-icon>
                      </template>
                    </n-popover>
                  </div>
                </div>
              </n-card>
            </div>
          </div>
          <div v-if="item.showType === 'file'">
            <n-tag class="my-2" effect="light" size="large" hit>
              {{ item.showName }}
            </n-tag>
            <n-table
              :data="item.fileList"
              :fit="true"
              border
              :header-cell-style="{ backgroundColor: '#f5f7fa' }"
            >
              <n-table-column
                v-for="col in item.colomn"
                :key="col.prop"
                :prop="col.prop"
                :label="col.label"
              >
                <template #default="scope">
                  <div v-if="col.prop === 'message'">
                    <span class="text-red-5">
                      {{ scope.row[col.prop] }}
                    </span>
                  </div>
                  <div v-else-if="col.prop === 'warn'">
                    <span class="text-yellow-5">
                      {{ scope.row[col.prop] }}
                    </span>
                  </div>
                  <div v-else>
                    <!-- 默认内容 -->
                    {{ scope.row[col.prop] }}
                  </div>
                </template>
              </n-table-column>
              <n-table-column label="操作">
                <template #default="scope">
                  <n-button size="small" @click="handleEdit(scope.$index, scope.row)">
                    下载
                  </n-button>
                </template>
              </n-table-column>
            </n-table>
          </div>
          <div v-if="item.showType === 'description'">
            <n-descriptions :column="2" border class="w-full">
              <n-descriptions-item
                v-for="des in item.descriptionList"
                :key="des.itemPid"
                :label="des.showName"
                label-class-name="w-1/5"
              >
                {{ des.itemValue }}
                <n-popover
                  v-if="des.resultType === '1' || des.resultType === '3'"
                  placement="right"
                  :width="200"
                  trigger="hover"
                  :content="des.message"
                >
                  <template #reference>
                    <n-icon class="text-red-4"><InfoFilled /></n-icon>
                  </template>
                </n-popover>
              </n-descriptions-item>
            </n-descriptions>
          </div>
          <div v-if="item.showType === 'table'" class="flex w-full mt-2">
            <n-table
              :data="item.tableList"
              :fit="true"
              border
              :header-cell-style="{ backgroundColor: '#f5f7fa' }"
            >
              <n-table-column
                v-for="col in item.colomn"
                :key="col.prop"
                :prop="col.prop"
                :label="col.label"
              >
                <template #default="scope">
                  <div v-if="col.prop === 'message'">
                    <span class="text-red-5">
                      {{ scope.row[col.prop] }}
                    </span>
                  </div>
                  <div v-else-if="col.prop === 'warn'">
                    <span class="text-yellow-5">
                      {{ scope.row[col.prop] }}
                    </span>
                  </div>
                  <div v-else>
                    <!-- 默认内容 -->
                    {{ scope.row[col.prop] }}
                  </div>
                </template>
              </n-table-column>
            </n-table>
          </div>
        </div>
      </div>
    </div>
    <n-backtop :right="100" :bottom="100" />
  </div>
</template>
<script lang="ts" setup>
  import { ref, onMounted, onBeforeUnmount } from 'vue';
  import { throttle } from 'lodash-es';
  import { getDetailDirectory, getPageDetail } from '~/services/cosmetics/index';
  import { PageDetailSection } from '~/services/cosmetics/type';
  import { useRoute } from 'vue-router';
  const route = useRoute();
  const dataInfo = JSON.parse(String(route.query.dataInfo));

  const selectedTagId = ref<string | null>(null);
  const scrollProgress = ref(0);
  const hzpContainer = ref(null);
  const isLoading = ref(true);
  const exampleData = ref<PageDetailSection[][]>([]);

  const handleEdit = (index: number, row: any) => {
    console.log(index, row);
  };
  const updateScrollProgress = () => {
    const container = hzpContainer.value as unknown as HTMLElement;
    if (!container) return;

    const scrollableHeight = container.scrollHeight - container.clientHeight;
    const currentScroll = container.scrollTop;
    const progress = (currentScroll / scrollableHeight) * 100;
    scrollProgress.value = Math.min(progress, 100);
    console.log(scrollProgress.value, '百分百');
  };
  const getDetailDirectoryData = async () => {
    console.log('123132112', dataInfo);
    try {
      let req = {
        ofVer: dataInfo.ofVer,
        functionId: '6111356347627266048',
        ofOrg: '',
        id: dataInfo.id,
        applyId: dataInfo.applyId,
      };
      const res = await getDetailDirectory(req);
      tagList.value = Array.isArray(res) ? res : [];
      console.log(tagList.value, 'tagList');
      isLoading.value = false;

      // 默认选中第一个项
      if (tagList.value.length > 0) {
        selectedTagId.value = tagList.value[0].id;
      }
      checkTagEmits(tagList.value[0]);
    } catch (error) {
      console.error('Error fetching data:', error);
    }
  };
  const checkTagEmits = async (tag: { id: string; functionId: string; moduleId: string }) => {
    isLoading.value = true;
    console.log(tag, 'tag');
    let reqdetail = {
      id: dataInfo.id,
      functionId: tag.functionId,
      moduleId: tag.moduleId,
      applyId: dataInfo.applyId,
    };
    selectedTagId.value = tag.id;
    try {
      const res3 = (await getPageDetail(reqdetail)) as unknown as PageDetailSection[][];
      exampleData.value = res3;
      console.log(exampleData.value, 'exampleData');
    } catch (error) {
      console.error('Error fetching page details:', error);
    } finally {
      isLoading.value = false;
    }
  };
  const tagList = ref<
    {
      id: string;
      functionId: string;
      moduleId: string;
      moduleName: string;
      messagesNum?: number;
    }[]
  >([
    {
      id: '',
      functionId: '',
      moduleId: '',
      moduleName: '',
    },
  ]);
  // 使用 lodash 节流函数
  const throttledUpdateScrollProgress = throttle(updateScrollProgress, 100);

  onMounted(() => {
    getDetailDirectoryData();

    const container = hzpContainer.value as unknown as HTMLElement;
    if (container) {
      container.addEventListener('scroll', throttledUpdateScrollProgress);
    }
  });

  onBeforeUnmount(() => {
    const container = hzpContainer.value as unknown as HTMLElement;
    if (container) {
      container.removeEventListener('scroll', throttledUpdateScrollProgress);
    }
  });
</script>
